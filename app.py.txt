def normalize(text: str) -> str:
    """기본적인 문자열 정규화"""
    text = text.lower()
    text = re.sub(r"\s+", " ", text)
    text = re.sub(r"[.,!?…]", "", text)
    return text.strip()


def contains_any(text: str, patterns: List[str]) -> bool:
    """후보 표현 중 하나라도 포함하는지 검사"""
    text = normalize(text)
    return any(normalize(p) in text for p in patterns)


def contains_all_groups(text: str, groups: List[List[str]]) -> bool:
    """
    각 그룹에서 최소 하나씩 발견되어야 True.
    예:
    [["혼자"], ["집중", "공부"]]
    """
    text = normalize(text)

    for group in groups:
        if not any(normalize(word) in text for word in group):
            return False

    return True

def detect_methods(answer: str) -> List[str]:
    """답안에 실제로 드러난 설명 방법 추정"""
    found = []

    for method, rule in EXPLANATION_METHODS.items():
        if contains_any(answer, rule["markers"]):
            found.append(method)

    return found


def extract_declared_method(answer: str) -> str:
    """
    문장 끝 괄호의 설명 방법 추출
    예: ...(비교와 대조)
    """
    pattern = r"\((정의|예시|인과|분석|비교와 대조|분류와 구분)\)"
    match = re.search(pattern, answer)

    if match:
        return match.group(1)

    return ""

        result.score += 1
        result.passed.append("㉡ 어려운 과제에서 혼자 집중하는 방법을 표현함")
    else:
        result.errors.append(
            "㉡은 어려운 과제에서 혼자 차분하게 집중하는 방법이 필요함"
        )

        result.score += 1
        result.passed.append("㉢ 사회적 억제를 정확히 제시함")

        result.errors.append(
            "㉢은 관련 심리 현상인 '사회적 억제'가 필요함"

        )

    return result

# =========================================================

def grade_explanation_pair(
    ans1,
    ans2,
    required_concepts_1,
    required_concepts_2,
    misconception_rules=None
):

    result = GradeResult(0, 5, "채점 중")

    # -----------------------
    # 1. 내용 검사
    # -----------------------

    if any(contains_any(ans1, group) for group in required_concepts_1):
        result.score += 1
        result.passed.append("(1)의 핵심 내용이 확인됨")
    else:
        result.errors.append("(1)에 요구된 핵심 내용이 충분히 드러나지 않음")

    if any(contains_any(ans2, group) for group in required_concepts_2):
        result.score += 1
        result.passed.append("(2)의 핵심 내용이 확인됨")
    else:
        result.errors.append("(2)에 요구된 핵심 내용이 충분히 드러나지 않음")

    # -----------------------
    # 2. 설명 방법 명칭
    # -----------------------

    declared1 = extract_declared_method(ans1)
    declared2 = extract_declared_method(ans2)

    if declared1:
        result.score += 1
        result.passed.append(f"(1) 설명 방법 표기: {declared1}")
    else:
        result.errors.append("(1) 문장 끝에 설명 방법 명칭이 없음")

    if declared2:
        result.score += 1
        result.passed.append(f"(2) 설명 방법 표기: {declared2}")
    else:
        result.errors.append("(2) 문장 끝에 설명 방법 명칭이 없음")

    # -----------------------
    # 3. 서로 다른 방법
    # -----------------------

    if declared1 and declared2 and declared1 != declared2:
        result.score += 1
        result.passed.append("서로 다른 설명 방법을 선택함")
    else:
        result.errors.append("두 문장의 설명 방법은 서로 달라야 함")

    # -----------------------
    # 4. 실제 설명 방법 구현 검사
    # -----------------------

    for idx, (answer, declared) in enumerate(
        [(ans1, declared1), (ans2, declared2)],
        start=1
    ):

        if declared:
            detected = detect_methods(answer)
            result.detected_methods.extend(detected)

            if declared not in detected:
                result.warnings.append(
                    f"({idx}) '{declared}'라고 표기했지만 "
                    f"해당 설명 방법의 특징이 문장에 명확히 드러나지 않을 수 있음"
                )

    # -----------------------
    # 5. 오개념 검사
    # -----------------------

    if misconception_rules:

        for rule in misconception_rules:

            if rule["condition"](ans1, ans2):
                result.errors.append(rule["message"])

    # -----------------------
    # 최종 상태
    # -----------------------

    if result.errors:
        result.status = "보완 필요"
    elif result.score >= 4:
        result.status = "통과"
    else:
        result.status = "부분 인정"

    return result

        )

    # ㉢
    value_found = (
        contains_any(ans3, SEMANTIC["art_world_change"])
        or contains_any(ans3, SEMANTIC["expand_art"])
        or contains_any(ans3, SEMANTIC["symbolic_value"])
    )

    if value_found:
        result.score += 1
        result.passed.append("㉢ AI 그림의 의미 또는 가치를 올바르게 제시함")

        result.errors.append(

    # 오개념
    if contains_any(ans2, SEMANTIC["no_value"]):
        result.errors.append(
            "오개념: AI 그림의 예술적 가치가 전혀 없다고 단정함"
        )

    result.status = "통과" if result.score == 3 and not result.errors else "보완 필요"

    return result

    if any(contains_any(visual, group) for group in visual_concepts):
        result.score += 1
        result.passed.append("시각 요소가 핵심 내용을 반영함")
    else:
        result.errors.append("시각 요소에 요구된 핵심 개념이 충분히 드러나지 않음")

    # 시각 효과
    if any(
        contains_any(visual_effect, group)
        for group in visual_effect_concepts
    ):
        result.score += 1
        result.passed.append("시각 요소의 효과가 지문 내용과 연결됨")
    else:
        result.errors.append("시각 요소의 효과에 지문 근거가 부족함")

    # 청각 요소
    if any(contains_any(audio, group) for group in audio_concepts):
        result.score += 1
        result.passed.append("청각 요소가 핵심 분위기와 연결됨")
    else:
        result.errors.append("청각 요소가 요구된 특성을 충분히 반영하지 못함")

    # 청각 효과
    if any(
        contains_any(audio_effect, group)
        for group in audio_effect_concepts
    ):
        result.score += 1
        result.passed.append("청각 요소의 효과가 지문 내용과 연결됨")
    else:
        result.errors.append("청각 요소의 효과에 지문 근거가 부족함")

    # 오개념 검사
    if misconception_patterns:
        combined = " ".join(
            [visual, visual_effect, audio, audio_effect]
        )

        for pattern, message in misconception_patterns:
            if contains_any(combined, pattern):
                result.errors.append(message)

    result.status = "통과" if result.score == 4 and not result.errors else "보완 필요"

    return result

def show_result(result: GradeResult):

    col1, col2 = st.columns([1, 3])

    with col1:
        st.metric(
            "점수",
            f"{result.score} / {result.max_score}"
        )

    with col2:
        if result.status == "통과":
            st.success(f"판정: {result.status}")
        elif result.status == "부분 인정":
            st.warning(f"판정: {result.status}")
        else:
            st.error(f"판정: {result.status}")

    if result.passed:
        st.subheader("✅ 충족한 조건")
        for item in result.passed:
            st.write("•", item)

    if result.warnings:
        st.subheader("⚠️ 확인이 필요한 부분")
        for item in result.warnings:
            st.write("•", item)

    if result.errors:
        st.subheader("❌ 보완이 필요한 부분")
        for item in result.errors:
            st.write("•", item)

    if result.detected_methods:
        st.subheader("🔎 감지된 설명 방법")
        st.write(", ".join(sorted(set(result.detected_methods))))
